       ```typescript
         surahNumber: number;
         ayahNumber: number;
         arabicText: string;
         translationText: string;
         edition: string;
       }
       ```

    2. Create src/store/useAppStore.ts with the full Zustand v5 store using the exact code from RESEARCH.md Pattern 4. The store MUST have:
       - VerseSelectionSlice: surahNumber (default 1), startAyah (default 1), endAyah (default 1), ayahs (default []), and setters. setSurah resets startAyah/endAyah to 1 and ayahs to [].
       - FontConfigSlice: arabicFontFamily (default 'Amiri Quran'), arabicFontSize (default 36), translationFontFamily (default 'Inter'), translationFontSize (default 18), and setters.
       - ColorConfigSlice: arabicTextColor (default '#FFFFFF'), translationTextColor (default '#FFFFFF'), and setters.
       - CanvasConfigSlice: aspectRatio (default '16:9', type '16:9' | '1:1' | '9:16'), showTranslation (default true), selectedTranslation (default 'eng-sahihinter'), and setters.
       - ExportStateSlice: fontsReady (default false), isExporting (default false), exportProgress (default 0), and setters.
       - Combined AppStore type = union of all slices.
       - Use `create<AppStore>()(...)` with double parentheses (Zustand v5 TypeScript requirement).
       - Import AyahData from '@/types/quran'.

    3. Create src/lib/utils/quranUtils.ts:
       ```typescript
       /**
        * Islamic requirement: Surah 9 (At-Tawbah) has NO Bismillah.
        * This is the only surah without one.
        * ALL other surahs (including Surah 1, Al-Fatiha) display Bismillah.
        */
       const SURAH_WITHOUT_BISMILLAH = 9;

       export function shouldShowBismillah(surahNumber: number): boolean {
         return surahNumber !== SURAH_WITHOUT_BISMILLAH;
       }
       ```

    4. Create src/lib/api/quranApi.ts as a stub for Phase 2:

       ```typescript
       /**
        * Quran API client — fetches verse data from fawazahmed0/quran-api via jsDelivr CDN.
        * Stub in Phase 1. Fleshed out in Phase 2.
        */

       const BASE_URL = 'https://cdn.jsdelivr.net/gh/fawazahmed0/quran-api@1';

       export const EDITIONS = {
         arabic: 'ara-quranuthmanihaf',
         translations: {
           'eng-sahihinter': 'Sahih International',
           'eng-clearquran': 'The Clear Quran (Dr. Khattab)',
         },
       } as const;

       export type TranslationEdition = keyof typeof EDITIONS.translations;
       ```
